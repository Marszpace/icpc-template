\section{Math}
	\subsection{Flows with lower bounds}
		% Used in ICPC 2026 Central, problem K (min flow on a DAG with edge and node bounds).
		Edge $u \to v$ with bounds $[l, c]$: give it capacity $c - l$ and set $\mathrm{ex}[v] \mathrel{+}= l$, $\mathrm{ex}[u] \mathrel{-}= l$.
		A node bounded by $[A, B]$ is split into $v_{in} \to v_{out}$ with bounds $[A, B]$.
		Add $S^* \to v$ of capacity $\mathrm{ex}[v]$ if $\mathrm{ex}[v] > 0$, and $v \to T^*$ of capacity $-\mathrm{ex}[v]$ if $\mathrm{ex}[v] < 0$.
		A feasible flow exists iff the max flow $S^* \to T^*$ saturates all edges out of $S^*$.
		For an $s$--$t$ flow also add $t \to s$ of capacity $\infty$; its flow $F$ is the flow value.
		Then delete it: \emph{max} flow is $F + \mathrm{maxflow}(s \to t)$ and \emph{min} flow is $F - \mathrm{maxflow}(t \to s)$, both in the residual graph.
		Paths covering a DAG's vertices: bound every vertex by $[1, \infty]$, connect $s$ to all sources and all sinks to $t$, take the min flow.
		Convex edge costs (e.g.\ $x^2$) in min-cost flow: use parallel unit edges with costs $1, 3, 5, \dots$

	\subsection{Min cut models}
		Max-weight closure (choose items, each needs its prerequisites): $s \to i$ with capacity $w_i$ if $w_i > 0$, $i \to t$ with $-w_i$ if $w_i < 0$,
		$i \to j$ with $\infty$ if $i$ needs $j$; answer $= \sum_{w_i > 0} w_i - \mathrm{mincut}$ (chosen items are on the $s$ side).
		Bipartite graph: max matching = min vertex cover (K\"onig), max independent set $= V -$ matching.
		With weights, min weight vertex cover is the min cut of $s \to L$ ($w$), $L \to R$ ($\infty$), $R \to t$ ($w$).
		DAG min path cover (vertex-disjoint) $= N -$ matching in the split graph (edge $u_{out} \to v_{in}$); paths may share vertices: use the transitive closure.
		Dilworth: max antichain $=$ min chain cover (the latter with the transitive closure).
		Edge-disjoint paths: unit capacities; vertex-disjoint: split $v$ into $v_{in} \to v_{out}$ of capacity 1.
		Two labels per item with pairwise penalties: $s$ side / $t$ side is the label; penalty edges between neighbours (needs submodular costs).

	\subsection{Segment tree graph}
		% Used in ICPC 2026 South, problems G and I.
		To add edges $v \to [l, r]$ cheaply, build a segment tree whose nodes are graph vertices, with edges parent $\to$ child (cost 0), and connect $v$ to the $O(\log N)$ nodes covering $[l, r]$.
		For $[l, r] \to v$ use a second tree with the same leaves and edges child $\to$ parent.
		This gives $O(N + M \log N)$ edges for Dijkstra, SCC, topological sort or BFS.

	\subsection{Erdős–Gallai theorem}
		% Source: https://en.wikipedia.org/wiki/Erd%C5%91s%E2%80%93Gallai_theorem
		% Test: stress-tests/graph/erdos-gallai.cpp
		A simple graph with node degrees $d_1 \ge \dots \ge d_n$ exists iff $d_1 + \dots + d_n$ is even and for every $k = 1\dots n$,
		\[ \sum _{i=1}^{k}d_{i}\leq k(k-1)+\sum _{i=k+1}^{n}\min(d_{i},k). \]
